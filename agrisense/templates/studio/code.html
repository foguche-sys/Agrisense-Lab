Switch the fan from Arduino (ESP32)

#include <WiFi.h>
#include <HTTPClient.h>
#include "secrets.h"      // WIFI_SSID, WIFI_PASS, TEAM_KEY

void setFan(const char* value) {
  HTTPClient http;
  http.begin("{{ base }}/api/v1/feeds/{{ team.slug }}.fan/data");
  http.addHeader("Content-Type", "application/json");
  http.addHeader("X-Team-Key", TEAM_KEY);
  int code = http.POST(String("{\"value\":\"") + value + "\"}");
  Serial.printf("fan %s -> HTTP %d\n", value, code);
  http.end();
}
